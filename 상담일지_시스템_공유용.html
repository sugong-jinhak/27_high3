// ============================================= // 📋 상담일지 시스템 - Google Apps Script 코드 (보안 수정판, 2026-11) // ============================================= // 바뀐 점 // - 모든 요청은 POST(JSON 본문)로만 받고, 매 요청마다 서버가 암호를 확인합니다. // - 진로교사 암호: 전체 열람·저장·삭제·설정 / 반 암호: 그 반 학생만. // - 암호는 어떤 응답에도 들어가지 않습니다. 새로 정한 암호는 해시로 저장합니다. // - 같은 대상(진로교사 또는 한 반)에 암호를 5번 틀리면 10분 잠금. // - 학생 기록은 여러 칸(A1, A2, ...)에 나눠 저장합니다. 예전 A1 한 칸 데이터도 그대로 읽습니다. // - 시트 메뉴 [상담일지 관리]에서 진로교사 암호 재설정, 잠금 풀기를 할 수 있습니다. // // 배포: [배포] > [배포 관리] > 연필(수정) > 버전: 새 버전 > [배포] // 실행 계정: 나 / 액세스 권한: 모든 사용자 (주소는 그대로 유지됩니다) // ============================================= const MIN_PW_LENGTH = 6; // 새 암호 최소 길이 const MAX_FAIL = 5; // 연속으로 틀릴 수 있는 횟수 const LOCK_SECONDS = 600; // 잠금 시간(초) = 10분 const CHUNK_SIZE = 40000; // 한 칸에 저장할 글자 수 (구글 시트 한 칸 최대 5만 자) const CHUNK_MARK = '§'; // 나눠 저장한 칸 앞에 붙이는 표시 문자 // ---------- 입구 ---------- function doGet() { return json_({ error: '이 주소는 상담일지 화면에서만 사용합니다.' }); } function doPost(e) { let req; try { req = JSON.parse((e && e.postData && e.postData.contents) || '{}'); } catch (err) { return json_({ error: '요청 형식이 올바르지 않습니다.' }); } const lock = LockService.getScriptLock(); if (!lock.tryLock(15000)) { return json_({ error: '다른 선생님이 저장 중입니다. 잠시 후 다시 시도해 주세요.' }); } try { return json_(handle_(req)); } catch (err) { return json_({ error: err.message || String(err), code: err.code || '' }); } finally { lock.releaseLock(); } } function handle_(req) { const action = req.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); const cfg = readConfig_(ss); // 로그인 화면용 공개 정보 (암호 없음) if (action === 'public') return { public: publicInfo_(cfg) }; const user = authenticate_(cfg, req.auth); if (action === 'load') { const all = readStudents_(ss); const list = user.role === 'career' ? all : all.filter(s => s && s.cls === user.cls); return { user: user, students: list, config: clientConfig_(cfg, user) }; } if (action === 'save') { const st = req.student; if (!st || typeof st !== 'object' || !st.id) fail_('학생 정보가 올바르지 않습니다.'); st.id = String(st.id); const all = readStudents_(ss); const idx = all.findIndex(s => s && s.id === st.id); if (user.role !== 'career') { if (idx >= 0 && all[idx].cls !== user.cls) fail_('다른 반 학생의 기록은 저장할 수 없습니다.', 'DENY'); if (st.cls !== user.cls) fail_('학생을 다른 반으로 옮겨 저장할 수 없습니다.', 'DENY'); } if (idx >= 0) all[idx] = st; else all.push(st); writeStudents_(ss, all); return { ok: true }; } if (action === 'delete') { const id = String(req.id || ''); const all = readStudents_(ss); const target = all.find(s => s && s.id === id); if (!target) fail_('삭제할 학생을 찾지 못했습니다.'); if (user.role !== 'career' && target.cls !== user.cls) fail_('다른 반 학생은 삭제할 수 없습니다.', 'DENY'); writeStudents_(ss, all.filter(s => !(s && s.id === id))); return { ok: true }; } // ----- 여기부터 진로교사만 ----- if (user.role !== 'career') fail_('진로교사만 할 수 있는 작업입니다.', 'DENY'); if (action === 'setConfig') { const inc = req.config || {}; if (Array.isArray(inc.years)) { const ys = inc.years.map(Number).filter(y => y >= 2000 && y <= 2100); cfg.years = ys.length ? Array.from(new Set(ys)) : [new Date().getFullYear()]; } if (inc.activeYear) cfg.activeYear = Number(inc.activeYear) || cfg.activeYear; if (inc.hr && typeof inc.hr === 'object') cfg.hr = inc.hr; if (inc.classInfo && typeof inc.classInfo === 'object') { Object.keys(inc.classInfo).forEach(k => { const v = inc.classInfo[k] || {}; const old = cfg.classInfo[k] || {}; const entry = { teacherName: String(v.teacherName != null ? v.teacherName : (old.teacherName || '')), pw: old.pw || '' // 기존 암호 유지 }; if (typeof v.newPw === 'string' && v.newPw) { // 새로 입력한 경우에만 변경 if (v.newPw.length < MIN_PW_LENGTH) fail_('반 암호는 ' + MIN_PW_LENGTH + '자 이상으로 정해 주세요.'); entry.pw = makePw_(v.newPw); clearFail_('cls_' + k); } cfg.classInfo[k] = entry; }); } writeConfig_(ss, cfg); return { ok: true, config: clientConfig_(cfg, user) }; } if (action === 'changeCareerPw') { const pw = String(req.newPw || ''); if (pw.length < MIN_PW_LENGTH) fail_('진로교사 암호는 ' + MIN_PW_LENGTH + '자 이상으로 정해 주세요.'); cfg.careerPw = makePw_(pw); writeConfig_(ss, cfg); clearFail_('career'); return { ok: true }; } if (action === 'importStudents') { const inc = Array.isArray(req.students) ? req.students : []; const all = readStudents_(ss); const ids = new Set(all.map(s => s && s.id)); let added = 0; inc.forEach(s => { if (s && typeof s === 'object' && s.id && !ids.has(String(s.id))) { s.id = String(s.id); all.push(s); ids.add(s.id); added++; } }); if (added) writeStudents_(ss, all); return { ok: true, added: added }; } fail_('알 수 없는 작업입니다.'); } // ---------- 암호 확인 · 잠금 ---------- function authenticate_(cfg, auth) { if (!auth || typeof auth.pw !== 'string' || !auth.pw) fail_('암호를 입력해 주세요.', 'AUTH'); if (auth.role === 'career') { checkLocked_('career'); if (!cfg.careerPw) fail_('진로교사 암호가 아직 없습니다. 구글 시트 메뉴 [상담일지 관리]에서 정해 주세요.', 'AUTH'); if (!pwMatch_(auth.pw, cfg.careerPw)) { recordFail_('career'); fail_('암호가 올바르지 않습니다.', 'AUTH'); } clearFail_('career'); return { role: 'career' }; } if (auth.role === 'homeroom') { const cls = String(auth.cls || '').slice(0, 100); const info = cfg.classInfo[cls]; checkLocked_('cls_' + cls); if (!info || !info.pw) fail_('해당 반의 암호가 설정되지 않았습니다.', 'AUTH'); if (!pwMatch_(auth.pw, info.pw)) { recordFail_('cls_' + cls); fail_('암호가 올바르지 않습니다.', 'AUTH'); } clearFail_('cls_' + cls); return { role: 'homeroom', cls: cls, name: info.teacherName || '' }; } fail_('로그인 정보가 올바르지 않습니다.', 'AUTH'); } function checkLocked_(key) { if (CacheService.getScriptCache().get('lock_' + key)) { fail_('암호를 ' + MAX_FAIL + '번 틀려 ' + Math.round(LOCK_SECONDS / 60) + '분 동안 잠겼습니다. 잠시 후 다시 시도해 주세요.', 'AUTH'); } } function recordFail_(key) { const c = CacheService.getScriptCache(); const n = Number(c.get('fail_' + key) || 0) + 1; c.put('fail_' + key, String(n), LOCK_SECONDS); if (n >= MAX_FAIL) { c.put('lock_' + key, '1', LOCK_SECONDS); c.remove('fail_' + key); } } function clearFail_(key) { CacheService.getScriptCache().removeAll(['fail_' + key, 'lock_' + key]); } // 새 암호는 "h1$소금$해시" 형태로 저장. 예전 평문 암호도 그대로 확인됨(새로 정하면 해시로 바뀜). function hashPw_(pw, salt) { const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, salt + pw, Utilities.Charset.UTF_8); return Utilities.base64Encode(bytes); } function makePw_(pw) { const salt = Utilities.getUuid().replace(/-/g, '').slice(0, 16); return 'h1$' + salt + '$' + hashPw_(pw, salt); } function pwMatch_(input, stored) { stored = String(stored == null ? '' : stored); if (!stored) return false; if (stored.indexOf('h1$') === 0) { const p = stored.split('$'); return p.length === 3 && hashPw_(input, p[1]) === p[2]; } return input === stored; } // ---------- 설정 ---------- function readConfig_(ss) { const sh = getOrCreate(ss, 'config'); const val = sh.getRange('A1').getValue(); let raw = {}; if (val) { try { raw = JSON.parse(val); } catch (err) { throw new Error('설정 데이터를 읽을 수 없습니다. (config 시트 A1 확인 필요)'); } } const y = new Date().getFullYear(); return { careerPw: raw.careerPw || '', years: Array.isArray(raw.years) && raw.years.length ? raw.years : [y], activeYear: parseInt(raw.activeYear) || y, hr: raw.hr && typeof raw.hr === 'object' ? raw.hr : {}, classInfo: raw.classInfo && typeof raw.classInfo === 'object' ? raw.classInfo : {} }; } function writeConfig_(ss, cfg) { const sh = getOrCreate(ss, 'config'); sh.getRange('A1').setNumberFormat('@').setValue(JSON.stringify(cfg)); } // 로그인 화면용: 학과·학년·반·담임 이름·연도만 function publicInfo_(cfg) { const classes = {}; Object.keys(cfg.classInfo).forEach(k => { classes[k] = { teacherName: cfg.classInfo[k].teacherName || '' }; }); return { years: cfg.years, activeYear: cfg.activeYear, hr: cfg.hr, classes: classes }; } // 로그인 후 화면에 보내는 설정: 암호 대신 "설정됨" 여부만 function clientConfig_(cfg, user) { const ci = {}; Object.keys(cfg.classInfo).forEach(k => { if (user.role === 'career' || k === user.cls) { ci[k] = { teacherName: cfg.classInfo[k].teacherName || '' }; if (user.role === 'career') ci[k].hasPw = !!cfg.classInfo[k].pw; } }); return { years: cfg.years, activeYear: cfg.activeYear, hr: cfg.hr, classInfo: ci }; } // ---------- 학생 기록 (여러 칸 나눠 저장) ---------- function readStudents_(ss) { const sh = getOrCreate(ss, 'students'); const last = sh.getLastRow(); if (last < 1) return []; const vals = sh.getRange(1, 1, last, 1).getValues().map(r => String(r[0] == null ? '' : r[0])); let text; if (vals[0].charAt(0) === CHUNK_MARK) { text = vals.filter(v => v.charAt(0) === CHUNK_MARK).map(v => v.slice(1)).join(''); } else { text = vals[0]; // 예전 방식: A1 한 칸 } if (!text) return []; try { const list = JSON.parse(text); return Array.isArray(list) ? list : []; } catch (err) { // 읽기에 실패하면 저장도 하지 않도록 오류로 멈춤 (기존 데이터 보호) throw new Error('학생 데이터를 읽을 수 없어 작업을 멈췄습니다. (students 시트 확인 필요)'); } } function writeStudents_(ss, list) { const sh = getOrCreate(ss, 'students'); const text = JSON.stringify(list); const rows = []; let i = 0; while (i < text.length) { let end = Math.min(i + CHUNK_SIZE, text.length); // 이모지 등 두 글자로 된 문자가 칸 경계에서 쪼개지지 않게 const code = text.charCodeAt(end - 1); if (end < text.length && code >= 0xD800 && code <= 0xDBFF) end--; rows.push([CHUNK_MARK + text.slice(i, end)]); i = end; } if (!rows.length) rows.push([CHUNK_MARK + '[]']); const oldLast = sh.getLastRow(); if (sh.getMaxRows() < rows.length) sh.insertRowsAfter(sh.getMaxRows(), rows.length - sh.getMaxRows()); sh.getRange(1, 1, rows.length, 1).setNumberFormat('@').setValues(rows); if (oldLast > rows.length) sh.getRange(rows.length + 1, 1, oldLast - rows.length, 1).clearContent(); SpreadsheetApp.flush(); // 저장 확인: 다시 읽어서 같은지 비교 const back = sh.getRange(1, 1, rows.length, 1).getValues().map(r => String(r[0]).slice(1)).join(''); if (back !== text) throw new Error('저장 확인에 실패했습니다. 다시 시도해 주세요.'); } // ---------- 공통 ---------- function getOrCreate(ss, name) { return ss.getSheetByName(name) || ss.insertSheet(name); } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function fail_(msg, code) { const e = new Error(msg); e.code = code || ''; throw e; } // ---------- 구글 시트 메뉴 (시트 편집 권한이 있는 사람만 보임) ---------- function onOpen() { SpreadsheetApp.getUi() .createMenu('상담일지 관리') .addItem('진로교사 암호 새로 정하기', 'menuSetCareerPw') .addItem('로그인 잠금 모두 풀기', 'menuClearLocks') .addSeparator() .addItem('반 암호 모두 지우기', 'menuClearClassPws') .addToUi(); } function menuSetCareerPw() { const ui = SpreadsheetApp.getUi(); const r = ui.prompt('진로교사 암호 새로 정하기', '새 암호를 입력하세요 (' + MIN_PW_LENGTH + '자 이상). 입력 내용이 화면에 보이니 주변을 확인하세요.', ui.ButtonSet.OK_CANCEL); if (r.getSelectedButton() !== ui.Button.OK) return; const pw = r.getResponseText(); if (pw.length < MIN_PW_LENGTH) { ui.alert(MIN_PW_LENGTH + '자 이상으로 입력해 주세요.'); return; } const lock = LockService.getScriptLock(); lock.waitLock(15000); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const cfg = readConfig_(ss); cfg.careerPw = makePw_(pw); writeConfig_(ss, cfg); clearFail_('career'); } finally { lock.releaseLock(); } ui.alert('진로교사 암호를 바꿨습니다.'); } function menuClearLocks() { const cfg = readConfig_(SpreadsheetApp.getActiveSpreadsheet()); const keys = ['fail_career', 'lock_career']; Object.keys(cfg.classInfo).forEach(k => keys.push('fail_cls_' + k, 'lock_cls_' + k)); CacheService.getScriptCache().removeAll(keys); SpreadsheetApp.getUi().alert('로그인 잠금을 모두 풀었습니다.'); } // 모든 반의 암호를 지움 (담임 이름·학반 목록·학생 기록은 그대로) // 지운 뒤에는 환경 설정 > 학반 관리에서 반마다 새 암호를 정해야 그 반 담임이 로그인할 수 있음 function menuClearClassPws() { const ui = SpreadsheetApp.getUi(); const ok = ui.alert('반 암호 모두 지우기', '모든 반의 암호를 지웁니다.\n지운 뒤에는 환경 설정에서 새 암호를 정하기 전까지 담임선생님들이 로그인할 수 없습니다.\n(담임 이름, 학반 목록, 학생 기록은 그대로입니다)\n\n정말 지우시겠습니까?', ui.ButtonSet.YES_NO); if (ok !== ui.Button.YES) return; const lock = LockService.getScriptLock(); lock.waitLock(15000); let n = 0; try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const cfg = readConfig_(ss); const keys = []; Object.keys(cfg.classInfo).forEach(k => { if (cfg.classInfo[k].pw) n++; cfg.classInfo[k].pw = ''; keys.push('fail_cls_' + k, 'lock_cls_' + k); }); writeConfig_(ss, cfg); CacheService.getScriptCache().removeAll(keys); } finally { lock.releaseLock(); } ui.alert('반 암호 ' + n + '개를 지웠습니다.\n진로교사로 로그인해 환경 설정 > 학반 관리에서 반마다 새 암호를 정해 주세요.'); }
